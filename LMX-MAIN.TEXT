procedure ProcessTheEvent; FORWARD;

function UserAborted : boolean;
{ TRUE if the user has pressed a key (or clicked) since we last looked.  }
{                                                                        }
{ A transfer runs INSIDE a command, so the main event loop is not        }
{ turning; without this the window is completely locked until the        }
{ timeout expires.  EventAvail/GetEvent let us peek at the queue from    }
{ here.  Only keyDown and buttonDown count as an abort; a Filer event    }
{ must NOT be swallowed here (an unanswered Filer op freezes the         }
{ desktop), so anything else is left in the queue for the main loop.     }
begin
   { EventAvail only says WHETHER an event is queued - it does not fill  }
   { theEvent.  The original version inspected the STALE previous event  }
   { and then blindly consumed the queued one: a swallowed folderUpdate  }
   { left the window blank, and a swallowed Filer event froze the        }
   { desktop.  Now: take the event, abort on a real key or click, and    }
   { hand anything else to the normal handler so updates, activates and  }
   { Filer traffic stay healthy even in mid-transfer.                    }
   UserAborted := false;
   if not EventAvail then
      exit(UserAborted);
   GetEvent(theEvent);
   if (theEvent.what = keyDown) or (theEvent.what = buttonDown) then begin
      cancelled := true;
      UserAborted := true
   end
   else
      ProcessTheEvent
end;

procedure DCp (code : integer; d0, d1, d2 : longint);
begin
   cparm.dccode := code;
   cparm.dcdata[0] := d0;
   cparm.dcdata[1] := d1;
   cparm.dcdata[2] := d2;
   DEVICE_CONTROL(ecode, pnp, cparm)
end;

procedure FlushIn;
begin
   DCp(9, -1, -2, -2)
end;

procedure SendB (v : integer);
begin
   if v > 127 then v := v - 256;
   sbuf[0] := v;
   WRITE_DATA(ecode, prefnum, ord4(@sbuf), 1, wrote, sequential, 0)
end;

function GetB (var bb : integer) : boolean;
var spins : longint;
    got   : boolean;
begin
   got := false;
   spins := 0;
   while (spins < POLLMAX) and (not got) and (not cancelled) do begin
      cparm.dccode := 19;
      DEVICE_CONTROL(ecode, pnp, cparm);
      if (ecode = 0) and (cparm.dcdata[9] > 0) then begin
         READ_DATA(ecode, prefnum, ord4(@rbuf), 1, actual,
                   sequential, 0);
         if (ecode = 0) and (actual = 1) then begin
            bb := rbuf[0];
            if bb < 0 then bb := bb + 256;
            got := true
         end
         else
            spins := POLLMAX
      end
      else begin
         spins := spins + 1;
         if (spins mod 64) = 0 then begin
            LetOthersRun;              { keep the desktop breathing }
            if UserAborted then
               spins := POLLMAX        { drop out of the wait at once }
         end
      end
   end;
   GetB := got
end;

function GetN (n : integer) : boolean;
{ Wait until n bytes are buffered, read them in ONE READ_DATA.           }
var spins : longint;
    got   : boolean;
begin
   got := false;
   spins := 0;
   while (spins < POLLMAX) and (not got) and (not cancelled) do begin
      cparm.dccode := 19;
      DEVICE_CONTROL(ecode, pnp, cparm);
      if (ecode = 0) and (cparm.dcdata[9] >= n) then begin
         READ_DATA(ecode, prefnum, ord4(@pbuf), n, actual,
                   sequential, 0);
         if (ecode = 0) and (actual = n) then
            got := true;
         spins := POLLMAX
      end
      else begin
         spins := spins + 1;
         if (spins mod 64) = 0 then begin
            LetOthersRun;
            if UserAborted then
               spins := POLLMAX
         end
      end
   end;
   GetN := got
end;

function PortUp : boolean;
{ MOUNT (1052 = already mounted = fine), configure, open.  The           }
{ Preferences assignment of Serial B persists on the boot volume.        }
{ If the terminal already opened the line, reuse it: closing and         }
{ reopening drops DTR, and a real modem hangs up on that.                }
begin
   if lineOpen then begin
      PortUp := true;
      exit(PortUp)
   end;
   PortUp := false;
   vol := '';
   pw  := '';
   dev := 'RS232B';
   MOUNT(ecode, vol, pw, dev);
   if (ecode <> 0) and (ecode <> 1052) then begin
      LmxWr('MOUNT ecode ');
      LmxNum(ecode);
      LmxWrLn('');
      if ecode = 1051 then
         SetupHint
   end;
   pnp := '-RS232B';
   cparm.dcversion := 2;
   DCp( 1, 0, 0, 0);
   DCp( 5, curBaud, 0, 0);
   DCp( 6, 1, 0, 0);
   DCp( 7, 0, 0, 0);
   DCp( 9, -1, -1, 32767);
   DCp( 9, 1024, 341, 683);            { 1.2 rev B: BACK to 1024.  The 8K }
                                        { experiment killed serial RX on   }
                                        { the real 2/10 - port came up     }
                                        { deaf, no modem banner.  Apple's  }
                                        { documented 1024 max is real on   }
                                        { this machine.  The menu-loss     }
                                        { problem needs a different fix.   }
   DCp(10, 0, 0, 0);
   DCp(14, 0, 0, 0);
   DCp(25, 0, 0, 0);
   DCp(11, 0, 0, 0);
   DCp(12, 0, 0, 0);
   DCp(17, 0, 0, 0);
   pnp := '-RS232B-XYZ';
   OPEN(ecode, pnp, prefnum, [DRead, DWrite]);
   pnp := '-RS232B';
   if ecode <= 0 then
      PortUp := true
   else begin
      LmxWr('port OPEN ecode ');
      LmxNum(ecode);
      LmxWrLn('');
      if (ecode = 946) or (ecode = 1051) then
         SetupHint
   end
end;

procedure Snooze (ms : integer);
{ 1.7: block this process for ms milliseconds (OS DELAY_TIME).  Used   }
{ for the Hayes guard time around +++.                                  }
var iv : timestmp_interval;
    ct : time_rec;
    e  : integer;
begin
   iv.sec := ms div 1000;
   iv.msec := ms mod 1000;
   ct.year := 0;
   ct.day := 1;
   ct.hour := 0;
   ct.minute := 0;
   ct.second := 0;
   ct.msec := 0;
   DELAY_TIME(e, iv, ct)
end;

procedure HangUp;
{ 1.7: hang up the MODEM, not just the port.  1.2 closed the serial     }
{ port and called it hung up; the WiRSa's session stayed connected and  }
{ the next ATDT went to the board as keystrokes (seen on the 2/10).     }
{ Hayes: a second of silence, +++, a second of silence, then ATH.       }
var b, n : integer;
    tmp  : string[1];
begin
   if lineOpen then begin
      LmxWrLn('hanging up: +++ then ATH (about three seconds) ...');
      Snooze(1200);
      SendB(ord('+'));
      SendB(ord('+'));
      SendB(ord('+'));
      Snooze(1200);
      SendB(ord('A'));
      SendB(ord('T'));
      SendB(ord('H'));
      SendB(13);
      Snooze(800);
      n := 0;                          { show what the modem said }
      while GetB(b) and (n < 200) do begin
         n := n + 1;
         if (b >= 32) and (b < 127) then begin
            tmp := ' ';
            tmp[1] := chr(b);
            LmxWr(tmp)
         end
         else if b = 13 then
            LmxWrLn('')
      end;
      LmxWrLn('');
      lineOpen := false;
      CLOSE_OBJECT(ecode, prefnum);
      LmxWrLn('line closed.')
   end
   else
      LmxWrLn('the line was not open.')
end;

procedure NumStr (n : longint; var s : Str255);
{ Lisa Pascal does not allow STRING function results, so this is a      }
{ VAR-parameter procedure.                                              }
var tmp : string[1];
    d   : integer;
begin
   if n <= 0 then
      s := '0'
   else begin
      s := '';
      while n > 0 do begin
         d := n mod 10;
         tmp := '0';
         tmp[1] := chr(ord('0') + d);
         s := concat(tmp, s);
         n := n div 10
      end
   end
end;

   if length(dotsLine) < 55 then
      dotsLine := concat(dotsLine, '.');

      curX := paneRect.left + StringWidth(paneBuf[paneCnt])
   end
end;

procedure HomePath (var nm : Pathname);   { 1.2b: was Str255 -> Error 125 }
{ 1.2: make a save name land on LISACom's own volume.  A leading '-'   }
{ means the user typed a full path: keep it.  A leading '#' is a full   }
{ path missing its hyphen (typed -#2#1- as #2#1-): add the hyphen.      }
{ Anything else is a bare name: prefix curVol (the working volume).     }
var tmp : string[1];
begin
   if length(nm) = 0 then
      exit(HomePath);
   if nm[1] = '#' then
      nm := concat('-', nm)
   else if nm[1] <> '-' then
      nm := concat(curVol, nm)           { 1.6: working volume }
end;

procedure DoRecvOp (var name : Str255);
{ cancelled is cleared here and tested throughout: any keypress during   }
{ the transfer aborts it cleanly instead of locking the window until     }
{ the timeout.                                                           }
var expected, prev  : integer;
    nblocks, errors : integer;
    b, blk, cmpl    : integer;
    chk, sum, d, i  : integer;
    done, fail, ok  : boolean;
    schedErr        : integer;
    bytes           : longint;
begin
   cancelled := false;
   { The Sched_Class bracket is held only around the file-system calls  }
   { (they can block on the disk); the transfer loop itself runs        }
   { preemptable, so the rest of the desktop stays healthy for the      }
   { whole minutes-long operation.                                      }
   fnm := name;
   HomePath(fnm);                      { 1.2: onto this volume }
   Sched_Class(schedErr, false);
   MAKE_FILE(ecode, fnm, 0);
   Sched_Class(schedErr, true);
   if (ecode > 0) and (ecode <> 890) then begin
      LmxWr('MAKE_FILE ecode ');
      LmxNum(ecode);
      LmxWrLn('');
      exit(DoRecvOp)
   end;
   Sched_Class(schedErr, false);
   OPEN(ecode, fnm, frefnum, [DWrite]);
   Sched_Class(schedErr, true);
   if ecode > 0 then begin
      LmxWr('file OPEN ecode ');
      LmxNum(ecode);
      LmxWrLn('');
      exit(DoRecvOp)
   end;
   if not PortUp then begin
      CLOSE_OBJECT(ecode, frefnum);
      exit(DoRecvOp)
   end;
   LmxWrLn('PC: File / Transfer / XMODEM / Send, Checksum');
   LmxWrLn('Waiting for the PC to start sending...');
   LmxWrLn('(syncing can take up to a minute - be patient)');
   FlushIn;
   expected := 1;
   nblocks := 0;
   errors := 0;
   dotsLine := '';
   done := false;
   fail := false;
   SendB(cNAK);
   while (not done) and (not fail) do begin
      if not GetB(b) then begin
         errors := errors + 1;
         if errors > 30 then fail := true
         else SendB(cNAK)
      end
      else if b = cEOT then begin
         SendB(cACK);
         done := true
      end
      else if b = cCAN then begin
         LmxWrLn('sender cancelled');
         fail := true
      end
      else if b = cSOH then begin
         ok := GetN(131);
         if ok then begin
            blk := pbuf[0];
            if blk < 0 then blk := blk + 256;
            cmpl := pbuf[1];
            if cmpl < 0 then cmpl := cmpl + 256;
            sum := 0;
            i := 0;
            while i < 128 do begin
               d := pbuf[i + 2];
               if d < 0 then d := d + 256;
               sum := (sum + d) mod 256;
               i := i + 1
            end;
            chk := pbuf[130];
            if chk < 0 then chk := chk + 256
         end;
         if not ok then begin
            errors := errors + 1;
            if errors > 10 then fail := true
            else begin FlushIn; SendB(cNAK) end
         end
         else if (blk + cmpl <> 255) or (chk <> sum) then begin
            errors := errors + 1;
            if errors > 10 then fail := true
            else begin FlushIn; SendB(cNAK) end
         end
         else if blk = expected then begin
            WRITE_DATA(ecode, frefnum, ord4(@pbuf) + 2, 128, wrote,
                       sequential, 0);
            if ecode <> 0 then begin
               LmxWr('file write ecode ');
               LmxNum(ecode);
               LmxWrLn('');
               fail := true
            end
            else begin
               errors := 0;
               nblocks := nblocks + 1;
               Progress(nblocks);
               if (nblocks mod 50) = 0 then begin
                  LmxWrLn('');
                  dotsLine := ''
               end;
               expected := (expected + 1) mod 256;
               SendB(cACK)
            end
         end
         else begin
            prev := expected - 1;
            if prev < 0 then prev := 255;
            if blk = prev then
               SendB(cACK)
            else begin
               errors := errors + 1;
               if errors > 10 then fail := true
               else begin FlushIn; SendB(cNAK) end
            end
         end
      end
   end;
   LmxWrLn('');
   if cancelled then begin
      SendB(cCAN);
      SendB(cCAN);
      LmxWrLn('RECEIVE CANCELLED')
   end
   else if fail then begin
      SendB(cCAN);
      SendB(cCAN);
      LmxWrLn('RECEIVE FAILED')
   end
   else begin
      Sched_Class(schedErr, false);
      TRUNCATE(ecode, frefnum);
      Sched_Class(schedErr, true);
      bytes := ord4(nblocks);
      bytes := bytes * 128;
      LmxWr('received ');
      LmxNum(nblocks);
      LmxWr(' blocks = ');
      LmxNum(bytes);
      LmxWrLn(' bytes')
   end;
   Sched_Class(schedErr, false);
   CLOSE_OBJECT(ecode, frefnum);
   Sched_Class(schedErr, true);
   ResumeHint;
   PortDn
end;

{------------------------------------------------------------------------}
{ YMODEM receive.  XMODEM's shape with three upgrades: CRC-16 instead   }
{ of the checksum, STX 1K blocks alongside SOH 128s, and a block 0      }
{ carrying the file's name and exact size.                               }
{------------------------------------------------------------------------}

procedure CrcByte (d : integer; var crc : longint);
{ Fold one byte into a CRC-16/XMODEM (poly $1021), bit-serially: Lisa   }
{ Pascal has no XOR, but XOR against the constant is three conditional  }
{ add/subtracts (bits $1000, $20, $01 - bit 0 is always 0 after the    }
{ shift, so the +1 is safe).                                             }
var k, mc, md : integer;
begin
   for k := 1 to 8 do begin
      if crc >= 32768 then mc := 1 else mc := 0;
      if d >= 128 then md := 1 else md := 0;
      crc := (crc * 2) mod 65536;
      d := (d * 2) mod 256;
      if mc <> md then begin
         if ((crc div 4096) mod 2) = 1 then
            crc := crc - 4096
         else
            crc := crc + 4096;
         if ((crc div 32) mod 2) = 1 then
            crc := crc - 32
         else
            crc := crc + 32;
         crc := crc + 1
      end
   end
end;

procedure CrcTabInit;
{ Build the nibble-XOR table and the split CRC table - runs once, at  }
{ the first YMODEM block.  Init-time arithmetic can be as slow as it  }
{ likes; only the per-byte path in GetBlkY has to be fast.             }
var i, a, b, k, r, w : integer;
    aa, bb           : integer;
    c                : longint;
begin
   for a := 0 to 15 do
      for b := 0 to 15 do begin
         aa := a;
         bb := b;
         r := 0;
         w := 8;
         for k := 1 to 4 do begin
            if (aa >= w) <> (bb >= w) then
               r := r + w;
            if aa >= w then aa := aa - w;
            if bb >= w then bb := bb - w;
            w := w div 2
         end;
         xor4[a * 16 + b] := r
      end;
   for i := 0 to 255 do begin
      c := 0;
      CrcByte(i, c);
      tabHH[i] := c div 4096;
      tabHL[i] := (c div 256) mod 16;
      tabLH[i] := (c mod 256) div 16;
      tabLL[i] := c mod 16
   end;
   crcTabOK := true
end;

function GetNAt (off, n : integer) : boolean;
{ Wait until n bytes are buffered, read them into pbuf at offset off.   }
{ The driver buffer is 1024 bytes, so a 1K block cannot be awaited      }
{ whole; callers fetch it in chunks of 512 or less.                      }
var spins : longint;
    got   : boolean;
begin
   got := false;
   spins := 0;
   while (spins < POLLMAX) and (not got) and (not cancelled) do begin
      cparm.dccode := 19;
      DEVICE_CONTROL(ecode, pnp, cparm);
      if (ecode = 0) and (cparm.dcdata[9] >= n) then begin
         READ_DATA(ecode, prefnum, ord4(@pbuf) + off, n, actual,
                   sequential, 0);
         if (ecode = 0) and (actual = n) then
            got := true;
         spins := POLLMAX
      end
      else begin
         spins := spins + 1;
         if (spins mod 64) = 0 then begin
            LetOthersRun;
            if UserAborted then
               spins := POLLMAX
         end
      end
   end;
   GetNAt := got
end;

function GetBlkY (var blen, blkno : integer) : integer;
{ One YMODEM block.  0 = good block in pbuf (header at 0..1, data from  }
{ 2, CRC verified); 1 = timeout or garbage; 2 = EOT; 3 = CAN.           }
var lead, i, d      : integer;
    rhi, rlo        : integer;
    cHH, cHL        : integer;
    cLH, cLL        : integer;
    dh, idx, nh, nl : integer;
    crcv            : longint;
    ok              : boolean;
begin
   GetBlkY := 1;
   if not GetB(lead) then
      exit(GetBlkY);
   if lead = cEOT then begin
      GetBlkY := 2;
      exit(GetBlkY)
   end;
   if lead = cCAN then begin
      GetBlkY := 3;
      exit(GetBlkY)
   end;
   if (lead <> cSOH) and (lead <> cSTX) then
      exit(GetBlkY);
   if lead = cSTX then blen := 1024 else blen := 128;
   { block number pair, then data + 2 CRC bytes, in chunks the driver  }
   { buffer can hold.                                                   }
   ok := GetNAt(0, 2);
   if ok then begin
      if blen = 1024 then begin
         ok := GetNAt(2, 512);
         if ok then ok := GetNAt(514, 512);
         if ok then ok := GetNAt(1026, 2)
      end
      else
         ok := GetNAt(2, 130)
   end;
   if not ok then
      exit(GetBlkY);
   blkno := pbuf[0];
   if blkno < 0 then blkno := blkno + 256;
   i := pbuf[1];
   if i < 0 then i := i + 256;
   if blkno + i <> 255 then
      exit(GetBlkY);
   if not crcTabOK then
      CrcTabInit;
   { crc' = (crc << 8) xor table[hi(crc) xor byte], done in nibbles.  }
   cHH := 0;
   cHL := 0;
   cLH := 0;
   cLL := 0;
   for i := 0 to blen - 1 do begin
      d := pbuf[i + 2];
      if d < 0 then d := d + 256;
      dh := d div 16;
      idx := xor4[cHH * 16 + dh] * 16
           + xor4[cHL * 16 + (d - dh * 16)];
      nh := xor4[cLH * 16 + tabHH[idx]];
      nl := xor4[cLL * 16 + tabHL[idx]];
      cLH := tabLH[idx];
      cLL := tabLL[idx];
      cHH := nh;
      cHL := nl
   end;
   crcv := ord4(cHH) * 4096 + cHL * 256 + cLH * 16 + cLL;
   rhi := pbuf[blen + 2];
   if rhi < 0 then rhi := rhi + 256;
   rlo := pbuf[blen + 3];
   if rlo < 0 then rlo := rlo + 256;
   if crcv <> (ord4(rhi) * 256 + rlo) then
      exit(GetBlkY);
   GetBlkY := 0
end;

procedure HdrParse (blen : integer; var nm : Str255; var sz : longint);
{ Block 0 data: filename, NUL, decimal size (then mtime etc - ignored). }
var i, d : integer;
    tmp  : string[1];
begin
   nm := '';
   sz := 0;
   i := 2;
   d := pbuf[i];
   if d < 0 then d := d + 256;
   while (d <> 0) and (i < blen + 2) and (length(nm) < 60) do begin
      if (d >= 32) and (d < 127) then begin
         tmp := ' ';
         tmp[1] := chr(d);
         nm := concat(nm, tmp)
      end;
      i := i + 1;
      d := pbuf[i];
      if d < 0 then d := d + 256
   end;
   i := i + 1;                         { past the NUL }
   while i < blen + 2 do begin
      d := pbuf[i];
      if d < 0 then d := d + 256;
      if (d >= ord('0')) and (d <= ord('9')) then begin
         sz := sz * 10 + (d - ord('0'));
         i := i + 1
      end
      else if (d = 32) and (sz = 0) then
         i := i + 1                    { leading spaces }
      else
         i := blen + 2                 { first non-digit ends the size }
   end
end;

procedure SanName (var nm : Str255);
{ The sender's filename, made safe for a Lisa catalog: path stripped,   }
{ odd characters become '_', length capped.                              }
{ Beta 23.  Two fixes here, both found the same evening.                 }
{ (1) '-' IS NOT LEGAL IN A LISA CATALOG NAME.  LOS uses it as the      }
{ pathname delimiter (-RS232B-, -#12-), so a hyphen anywhere in the     }
{ name makes MAKE_FILE fail: LMX-MAIN.TEXT off the wire returned ecode  }
{ 1282, while LMXMAIN.TEXT went through untouched.  Compare documented  }
{ 1023 "Invalid newname, check for '-' in string" and 1171 "'-' is not  }
{ allowed in a volume name".  Every substitution and every passed-      }
{ through hyphen now becomes '_' instead, which is known good.  This    }
{ bug predates the beta 23 work - any incoming name with a space or a   }
{ '+' already became a hyphen and already failed; it went unnoticed     }
{ because the name is usually typed at the prompt rather than taken     }
{ from the sender.                                                       }
{ (2) A catalog name takes only ONE period and LOS CUTS the name at the }
{ second, so T419306ICON.BACKUP.bin arrived as T419306ICON.BACKUP with  }
{ the real extension lost.  Periods after the first now fold to '_'     }
{ too, giving T419306ICON.BACKUP_bin - one period, still readable.      }
var i, k   : integer;
    out    : Str255;
    ch     : char;
    tmp    : string[1];
    gotDot : boolean;
begin
   k := 0;                             { last path separator }
   for i := 1 to length(nm) do
      if (nm[i] = '/') or (nm[i] = chr(92)) then k := i;
   out := '';
   gotDot := false;
   for i := k + 1 to length(nm) do begin
      ch := nm[i];
      if not (((ch >= 'A') and (ch <= 'Z')) or ((ch >= 'a') and (ch <= 'z'))
         or ((ch >= '0') and (ch <= '9')) or (ch = '.') or (ch = '_')) then
         ch := '_';
      if ch = '.' then begin
         if gotDot then
            ch := '_'
         else
            gotDot := true
      end;
      if length(out) < 30 then begin
         tmp := ' ';
         tmp[1] := ch;
         out := concat(out, tmp)
      end
   end;
   if length(out) = 0 then
      out := 'YMODEM.RECV';
   nm := out
end;

procedure DoRecvYOp (var name : Str255);
{ Beta 24: BATCH receive.  YMODEM sends any number of files in one      }
{ session; each gets its own block-0 header, and an EMPTY header ends   }
{ the batch.  The old code took the first file and CANned the rest;     }
{ now the whole per-file engine sits inside a loop that runs until the  }
{ empty header (or a failure) arrives.  A name typed at the prompt      }
{ applies to the FIRST file only; every later file takes the sender's   }
{ name through SanName.  Each file is truncated and closed as its own   }
{ EOT lands, so files already received survive a failure later in the  }
{ batch.  The inner header/data code is beta 22's proven engine,        }
{ unchanged; only the loop control around it is new.                    }
var expected, prev  : integer;
    nblocks, errors : integer;
    blen, blkno, r  : integer;
    done, fail      : boolean;
    gotHdr          : boolean;
    schedErr        : integer;
    eots            : integer;
    hdrName         : Str255;
    hdrSize         : longint;
    remaining       : longint;
    written         : longint;
    w               : longint;
    fileCnt         : integer;        { files completed this session }
    batchDn         : boolean;        { empty header seen - batch over }
begin
   cancelled := false;
   if not PortUp then
      exit(DoRecvYOp);
   LmxWrLn('YMODEM receive.  Start the send on the BBS (Ymodem,');
   LmxWrLn('not Ymodem-G), then this side syncs.  Any key aborts.');
   FlushIn;
   fileCnt := 0;
   batchDn := false;
   fail := false;
   while (not batchDn) and (not fail) do begin
   { ---- block 0: this file's header ---- }
   errors := 0;
   gotHdr := false;
   while (not gotHdr) and (not fail) do begin
      SendB(cCRC);
      r := GetBlkY(blen, blkno);
      if cancelled then fail := true
      else if (r = 0) and (blkno = 0) then begin
         HdrParse(blen, hdrName, hdrSize);
         SendB(cACK);
         gotHdr := true
      end
      else if r = 3 then begin
         LmxWrLn('sender cancelled');
         fail := true
      end
      else begin
         errors := errors + 1;
         if errors > 10 then fail := true
         else FlushIn
      end
   end;
   if fail then begin
      LmxWrLn('');
      if cancelled then LmxWrLn('RECEIVE CANCELLED')
      else LmxWrLn('no YMODEM header - RECEIVE FAILED')
   end
   else if length(hdrName) = 0 then begin
      { the empty header: end of batch }
      if fileCnt = 0 then
         LmxWrLn('the sender had nothing to send.');
      batchDn := true
   end
   else begin
   if (length(name) > 0) and (fileCnt = 0) then
      fnm := name
   else begin
      SanName(hdrName);
      fnm := hdrName
   end;
   HomePath(fnm);                      { 1.2: onto this volume }
   LmxWr('receiving into ');
   LmxWr(fnm);
   if hdrSize > 0 then begin
      LmxWr(', ');
      LmxNum(hdrSize);
      LmxWr(' bytes')
   end;
   LmxWrLn('');
   Sched_Class(schedErr, false);
   MAKE_FILE(ecode, fnm, 0);
   Sched_Class(schedErr, true);
   if (ecode > 0) and (ecode <> 890) then begin
      LmxWr('MAKE_FILE ecode ');
      LmxNum(ecode);
      LmxWrLn('');
      SendB(cCAN);
      SendB(cCAN);
      fail := true
   end;
   if not fail then begin
   Sched_Class(schedErr, false);
   OPEN(ecode, fnm, frefnum, [DWrite]);
   Sched_Class(schedErr, true);
   if ecode > 0 then begin
      LmxWr('file OPEN ecode ');
      LmxNum(ecode);
      LmxWrLn('');
      SendB(cCAN);
      SendB(cCAN);
      fail := true
   end
   end;
   if not fail then begin
   { ---- the data ---- }
   SendB(cCRC);                        { start the sender }
   expected := 1;
   nblocks := 0;
   errors := 0;
   eots := 0;
   dotsLine := '';
   remaining := hdrSize;
   written := 0;
   done := false;
   while (not done) and (not fail) do begin
      r := GetBlkY(blen, blkno);
      if cancelled then
         fail := true
      else if r = 2 then begin         { EOT: NAK the first, ACK the second }
         eots := eots + 1;
         if eots = 1 then
            SendB(cNAK)
         else begin
            SendB(cACK);
            done := true
         end
      end
      else if r = 3 then begin
         LmxWrLn('');
         LmxWrLn('sender cancelled');
         fail := true
      end
      else if r <> 0 then begin
         errors := errors + 1;
         if errors > 10 then fail := true
         else begin FlushIn; SendB(cNAK) end
      end
      else if blkno = expected then begin
         w := blen;
         if (hdrSize > 0) and (remaining < w) then
            w := remaining;
         if w > 0 then
            WRITE_DATA(ecode, frefnum, ord4(@pbuf) + 2, w, wrote,
                       sequential, 0)
         else
            ecode := 0;
         if ecode <> 0 then begin
            LmxWr('file write ecode ');
            LmxNum(ecode);
            LmxWrLn('');
            fail := true
         end
         else begin
            errors := 0;
            written := written + w;
            if hdrSize > 0 then
               remaining := remaining - w;
            nblocks := nblocks + 1;
            Progress(nblocks);
            if (nblocks mod 50) = 0 then begin
               LmxWrLn('');
               dotsLine := ''
            end;
            expected := (expected + 1) mod 256;
            SendB(cACK)
         end
      end
      else begin
         prev := expected - 1;
         if prev < 0 then prev := 255;
         if blkno = prev then
            SendB(cACK)                { a duplicate: already written }
         else begin
            errors := errors + 1;
            if errors > 10 then fail := true
            else begin FlushIn; SendB(cNAK) end
         end
      end
   end;
   LmxWrLn('');
   if cancelled then begin
      SendB(cCAN);
      SendB(cCAN);
      LmxWrLn('RECEIVE CANCELLED');
      fail := true
   end
   else if fail then begin
      SendB(cCAN);
      SendB(cCAN);
      LmxWrLn('RECEIVE FAILED')
   end
   else begin
      Sched_Class(schedErr, false);
      TRUNCATE(ecode, frefnum);
      Sched_Class(schedErr, true);
      if ecode <> 0 then begin
         LmxWr('TRUNCATE ecode ');
         LmxNum(ecode);
         LmxWrLn('')
      end;
      LmxWr('received ');
      LmxNum(written);
      LmxWr(' bytes into ');
      LmxWrLn(fnm);
      fileCnt := fileCnt + 1
   end;
   Sched_Class(schedErr, false);
   CLOSE_OBJECT(ecode, frefnum);
   Sched_Class(schedErr, true);
   if ecode <> 0 then begin
      LmxWr('CLOSE ecode ');
      LmxNum(ecode);
      LmxWrLn('')
   end
   end                                 { if not fail (file open) }
   end                                 { else: a real header }
   end;                                { while not batchDn }
   if fileCnt > 1 then begin
      LmxWr('batch complete: ');
      LmxNum(fileCnt);
      LmxWrLn(' files received')
   end;
   ResumeHint;
   PortDn
end;

procedure SrvInk (ch : char; var n : integer);
{ Same idea as the pane probe's ink test: draw ch in the current font  }
{ near the top left, count black pixels, erase.                        }
var s    : string[1];
    x, y : integer;
    box  : Rect;
begin
   box.left := 2;
   box.top := 2;
   box.right := 34;
   box.bottom := 30;
   EraseRect(box);
   MoveTo(4, 24);
   s := ' ';
   s[1] := ch;
   DrawString(s);
   n := 0;
   for y := 3 to 29 do
      for x := 3 to 33 do
         if GetPixel(x, y) then
            n := n + 1;
   EraseRect(box)
end;

procedure FontSurvey;
{ Walk font slots 1..60 and print one line for every fixed-width font  }
{ that draws real glyphs - the line itself drawn IN that font, so the  }
{ screen is its own specimen sheet.  Wider limits than the pane probe  }
{ (4..12 px cells, 6..24 px lines) so bigger strikes show up too.      }
{ The pane stays in its current font afterwards.                        }
var fid, w1, lh   : integer;
    inkI, inkM    : integer;
    fi            : FontInfo;
    hits          : integer;
begin
   LmxWrLn('');
   LmxWrLn('Font survey: fixed-width slots 1..60, each line in');
   LmxWrLn('its own font.  Note the number you like best.');
   hits := 0;
   for fid := 1 to 60 do begin
      SetPort(myWindow);
      TextFont(fid);
      GetFontInfo(fi);
      w1 := StringWidth('i');
      lh := fi.ascent + fi.descent + fi.leading;
      if (w1 = StringWidth('M')) and (w1 = StringWidth('.'))
         and (w1 >= 4) and (w1 <= 12)
         and (lh >= 6) and (lh <= 24) then begin
         SrvInk('i', inkI);
         SrvInk('M', inkM);
         if (inkI > 0) and (inkM > inkI + 4) then begin
            hits := hits + 1;
            { print the specimen line in the candidate font itself:   }
            { temporarily retarget the pane metrics so LmxWr draws in }
            { it at a sane baseline, then restore.                     }
            LmxWr('font ');
            LmxNum(fid);
            LmxWr(': ');
            LmxNum(w1);
            LmxWr('x');
            LmxNum(lh);
            LmxWr('  ABCdefgh 0123 iiiWWW');
            LmxWrLn('');
            TextFont(paneFont)      { back before the next line draws }
         end
      end
   end;
   SetPort(myWindow);
   if paneMono then TextFont(paneFont) else TextFont(0);
   LmxWr('survey done: ');
   LmxNum(hits);
   LmxWrLn(' fixed-width slot(s) found.');
   LmxWrLn('Tell the toolsmith the winning number.')
end;

procedure DoSendOp (var name : Str255);
var blk, nblocks, tries : integer;
    b, sum, d, i        : integer;
    done, fail, acked   : boolean;
    started             : boolean;
    hinted              : boolean;
    schedErr            : integer;
    bytes               : longint;
begin
   cancelled := false;
   fnm := name;
   HomePath(fnm);                      { 1.5: bare name = this volume }
   Sched_Class(schedErr, false);
   OPEN(ecode, fnm, frefnum, [DRead]);
   Sched_Class(schedErr, true);
   if ecode > 0 then begin
      LmxWr('file OPEN ecode ');
      LmxNum(ecode);
      LmxWrLn('');
      exit(DoSendOp)
   end;
   if not PortUp then begin
      CLOSE_OBJECT(ecode, frefnum);
      exit(DoSendOp)
   end;
   LmxWrLn('PC: File / Transfer / XMODEM / Receive, Checksum');
   LmxWrLn('Waiting for the PC receiver to answer...');
   LmxWrLn('(syncing can take up to a minute - be patient)');
   FlushIn;
   started := false;
   fail := false;
   hinted := false;
   tries := 0;
   while (not started) and (not fail) and (not cancelled) do begin
      if GetB(b) then begin
         if b = cNAK then started := true
         else if b = cCRC then begin
            { the receiver is alive, just asking for CRC.  Tera Term    }
            { falls back to checksum NAKs by itself after ~30 seconds,  }
            { so KEEP WAITING - a C must never count toward giving up.  }
            tries := 0;
            if not hinted then begin
               LmxWrLn('receiver wants CRC - waiting for its checksum');
               LmxWrLn('fallback (or set Checksum on the PC now)');
               hinted := true
            end
         end
         else if b = cCAN then begin
            LmxWrLn('the PC cancelled');
            fail := true
         end
      end
      else begin
         tries := tries + 1;
         if tries > 30 then begin
            LmxWrLn('no receiver answered - giving up');
            fail := true
         end
      end
   end;
   if cancelled then
      fail := true;
   blk := 1;
   nblocks := 0;
   dotsLine := '';
   done := false;
   while (not done) and (not fail) do begin
      READ_DATA(ecode, frefnum, ord4(@fbuf), 128, actual,
                sequential, 0);
      fcount := actual;
      if fcount <= 0 then
         done := true
      else begin
         i := 0;
         while i < 128 do begin
            if i >= fcount then fbuf[i] := 26;
            i := i + 1
         end;
         pkt[0] := cSOH;
         if blk > 127 then pkt[1] := blk - 256 else pkt[1] := blk;
         d := 255 - blk;
         if d > 127 then pkt[2] := d - 256 else pkt[2] := d;
         sum := 0;
         i := 0;
         while i < 128 do begin
            pkt[i + 3] := fbuf[i];
            d := fbuf[i];
            if d < 0 then d := d + 256;
            sum := (sum + d) mod 256;
            i := i + 1
         end;
         if sum > 127 then pkt[131] := sum - 256 else pkt[131] := sum;
         tries := 0;
         acked := false;
         while (not acked) and (not fail) and (not cancelled) do begin
            WRITE_DATA(ecode, prefnum, ord4(@pkt), 132, wrote,
                       sequential, 0);
            if ecode <> 0 then
               fail := true
            else if not GetB(b) then begin
               tries := tries + 1;
               if tries > 10 then fail := true
            end
            else if b = cACK then
               acked := true
            else if b = cCAN then begin
               LmxWrLn('receiver cancelled');
               fail := true
            end
            else begin
               tries := tries + 1;
               if tries > 10 then fail := true
            end
         end;
         if acked then begin
            nblocks := nblocks + 1;
            Progress(nblocks);
            if (nblocks mod 50) = 0 then begin
               LmxWrLn('');
               dotsLine := ''
            end;
            blk := (blk + 1) mod 256;
            if fcount < 128 then done := true
         end
      end
   end;
   if not fail then begin
      tries := 0;
      acked := false;
      while (not acked) and (tries <= 10) do begin
         SendB(cEOT);
         if GetB(b) then begin
            if b = cACK then acked := true
            else tries := tries + 1
         end
         else
            tries := tries + 1
      end
   end;
   LmxWrLn('');
   if cancelled then begin
      SendB(cCAN);
      SendB(cCAN);
      LmxWrLn('SEND CANCELLED')
   end
   else if fail then begin
      SendB(cCAN);
      SendB(cCAN);
      LmxWrLn('SEND FAILED')
   end
   else begin
      bytes := ord4(nblocks);
      bytes := bytes * 128;
      LmxWr('sent ');
      LmxNum(nblocks);
      LmxWr(' blocks = ');
      LmxNum(bytes);
      LmxWrLn(' bytes on the wire')
   end;
   Sched_Class(schedErr, false);
   CLOSE_OBJECT(ecode, frefnum);
   Sched_Class(schedErr, true);
   PortDn
end;

procedure YCrcPkt (n : integer);
{ CRC-16/XMODEM over pkt[3..n+2], stored big-endian at pkt[n+3..n+4].   }
{ CrcByte is bit-serial and slower than the receive tables; on a send   }
{ the wire wait dominates, so simple wins over fast here.                }
var i, d : integer;
    c    : longint;
begin
   c := 0;
   for i := 0 to n - 1 do begin
      d := pkt[i + 3];
      if d < 0 then d := d + 256;
      CrcByte(d, c)
   end;
   d := c div 256;
   if d > 127 then pkt[n + 3] := d - 256 else pkt[n + 3] := d;
   d := c mod 256;
   if d > 127 then pkt[n + 4] := d - 256 else pkt[n + 4] := d
end;

procedure DoSendYOp (var name : Str255);
{ 1.0: YMODEM send, Lisa to PC.  The XMODEM engine's shape with three   }
{ additions: a block-0 header carrying name and exact byte size (so     }
{ the receiver truncates the 0x1A tail padding away), CRC-16 on every   }
{ block via YCrcPkt, and the closing empty header that ends a YMODEM    }
{ session.  The file is read once first to count its exact size - the   }
{ Lisa's recorded length can include page padding, and the wire needs   }
{ the truth - then closed and reopened to rewind.  '/' in the Lisa      }
{ name becomes '_' for the PC's benefit.  PC side: Tera Term File /     }
{ Transfer / YMODEM / Receive.                                           }
var blk, nblocks, tries : integer;
    b, d, i, k          : integer;
    done, fail, acked   : boolean;
    started             : boolean;
    hinted              : boolean;
    schedErr            : integer;
    fsize               : longint;
    ns                  : Str255;
begin
   cancelled := false;
   fnm := name;
   HomePath(fnm);                      { 1.5: bare name = this volume }
   Sched_Class(schedErr, false);
   OPEN(ecode, fnm, frefnum, [DRead]);
   Sched_Class(schedErr, true);
   if ecode > 0 then begin
      LmxWr('file OPEN ecode ');
      LmxNum(ecode);
      LmxWrLn('');
      exit(DoSendYOp)
   end;
   if not PortUp then begin
      CLOSE_OBJECT(ecode, frefnum);
      exit(DoSendYOp)
   end;
   { ---- pass 1: count the exact bytes ---- }
   fsize := 0;
   actual := 1;
   while actual > 0 do begin
      READ_DATA(ecode, frefnum, ord4(@fbuf), 128, actual,
                sequential, 0);
      if actual > 0 then
         fsize := fsize + actual
   end;
   Sched_Class(schedErr, false);
   CLOSE_OBJECT(ecode, frefnum);
   OPEN(ecode, fnm, frefnum, [DRead]);
   Sched_Class(schedErr, true);
   if ecode > 0 then begin
      LmxWr('file reOPEN ecode ');
      LmxNum(ecode);
      LmxWrLn('');
      PortDn;
      exit(DoSendYOp)
   end;
   LmxWr('sending ');
   LmxWr(fnm);
   LmxWr(', ');
   LmxNum(fsize);
   LmxWrLn(' bytes');
   LmxWrLn('PC: File / Transfer / YMODEM / Receive');
   LmxWrLn('Waiting for the PC receiver...');
   FlushIn;
   if not crcTabOK then
      CrcTabInit;
   { ---- wait for the receiver's C ---- }
   started := false;
   fail := false;
   hinted := false;
   tries := 0;
   while (not started) and (not fail) and (not cancelled) do begin
      if GetB(b) then begin
         if b = cCRC then started := true
         else if b = cNAK then begin
            if not hinted then begin
               LmxWrLn('receiver is asking for XMODEM checksum -');
               LmxWrLn('pick YMODEM on the PC side');
               hinted := true
            end
         end
         else if b = cCAN then begin
            LmxWrLn('the PC cancelled');
            fail := true
         end
      end
      else begin
         tries := tries + 1;
         if tries > 30 then begin
            LmxWrLn('no receiver answered - giving up');
            fail := true
         end
      end
   end;
   if cancelled then
      fail := true;
   { ---- block 0: name, NUL, decimal size ---- }
   if not fail then begin
      pkt[0] := cSOH;
      pkt[1] := 0;
      pkt[2] := 255 - 256;              { complement of block number 0 }
      for i := 0 to 127 do
         pkt[i + 3] := 0;
      { 1.4: send the bare name -- skip a -vol- prefix (names cannot   }
      { contain '-', so the last hyphen ends the prefix).              }
      k := 0;
      if fnm[1] = '-' then
         for d := 2 to length(fnm) do
            if fnm[d] = '-' then k := d;
      i := 0;
      while (k + i < length(fnm)) and (i < 60) do begin
         d := ord(fnm[k + i + 1]);
         if d = ord('/') then d := ord('_');
         pkt[i + 3] := d;
         i := i + 1
      end;
      i := i + 1;                       { the NUL after the name }
      NumStr(fsize, ns);
      d := 1;
      while d <= length(ns) do begin
         pkt[i + 3] := ord(ns[d]);
         i := i + 1;
         d := d + 1
      end;
      YCrcPkt(128);
      tries := 0;
      acked := false;
      while (not acked) and (not fail) and (not cancelled) do begin
         WRITE_DATA(ecode, prefnum, ord4(@pkt), 133, wrote,
                    sequential, 0);
         if ecode <> 0 then
            fail := true
         else if not GetB(b) then begin
            tries := tries + 1;
            if tries > 10 then fail := true
         end
         else if b = cACK then
            acked := true
         else if b = cCAN then begin
            LmxWrLn('receiver cancelled');
            fail := true
         end
         else begin
            tries := tries + 1;
            if tries > 10 then fail := true
         end
      end;
      { after ACKing the header the receiver sends C for the data }
      if not fail then begin
         tries := 0;
         started := false;
         while (not started) and (not fail) and (not cancelled) do begin
            if GetB(b) then begin
               if b = cCRC then started := true
               else if b = cCAN then fail := true
            end
            else begin
               tries := tries + 1;
               if tries > 10 then fail := true
            end
         end
      end
   end;
   if cancelled then
      fail := true;
   { ---- the data, 128-byte CRC blocks ---- }
   blk := 1;
   nblocks := 0;
   dotsLine := '';
   done := false;
   while (not done) and (not fail) do begin
      READ_DATA(ecode, frefnum, ord4(@fbuf), 128, actual,
                sequential, 0);
      fcount := actual;
      if fcount <= 0 then
         done := true
      else begin
         i := 0;
         while i < 128 do begin
            if i >= fcount then fbuf[i] := 26;
            i := i + 1
         end;
         pkt[0] := cSOH;
         if blk > 127 then pkt[1] := blk - 256 else pkt[1] := blk;
         d := 255 - blk;
         if d > 127 then pkt[2] := d - 256 else pkt[2] := d;
         for i := 0 to 127 do
            pkt[i + 3] := fbuf[i];
         YCrcPkt(128);
         tries := 0;
         acked := false;
         while (not acked) and (not fail) and (not cancelled) do begin
            WRITE_DATA(ecode, prefnum, ord4(@pkt), 133, wrote,
                       sequential, 0);
            if ecode <> 0 then
               fail := true
            else if not GetB(b) then begin
               tries := tries + 1;
               if tries > 10 then fail := true
            end
            else if b = cACK then
               acked := true
            else if b = cCAN then begin
               LmxWrLn('receiver cancelled');
               fail := true
            end
            else begin
               tries := tries + 1;
               if tries > 10 then fail := true
            end
         end;
         if acked then begin
            nblocks := nblocks + 1;
            Progress(nblocks);
            if (nblocks mod 50) = 0 then begin
               LmxWrLn('');
               dotsLine := ''
            end;
            blk := (blk + 1) mod 256;
            if fcount < 128 then done := true
         end
      end
   end;
   if cancelled then
      fail := true;
   { ---- EOT, then the closing empty header ---- }
   if not fail then begin
      tries := 0;
      acked := false;
      while (not acked) and (tries <= 10) do begin
         SendB(cEOT);
         if GetB(b) then begin
            if b = cACK then acked := true
            else tries := tries + 1
         end
         else
            tries := tries + 1
      end;
      if acked then begin
         { the receiver now asks for the next file with C; the empty  }
         { header tells it there is none.                              }
         tries := 0;
         started := false;
         while (not started) and (tries <= 10) do begin
            if GetB(b) then begin
               if b = cCRC then started := true
               else tries := tries + 1
            end
            else
               tries := tries + 1
         end;
         if started then begin
            pkt[0] := cSOH;
            pkt[1] := 0;
            pkt[2] := 255 - 256;
            for i := 0 to 127 do
               pkt[i + 3] := 0;
            YCrcPkt(128);
            WRITE_DATA(ecode, prefnum, ord4(@pkt), 133, wrote,
                       sequential, 0);
            tries := 0;
            acked := false;
            while (not acked) and (tries <= 5) do begin
               if GetB(b) then begin
                  if b = cACK then acked := true
                  else tries := tries + 1
               end
               else
                  tries := 6
            end
         end
      end
   end;
   LmxWrLn('');
   if cancelled then begin
      SendB(cCAN);
      SendB(cCAN);
      LmxWrLn('SEND CANCELLED')
   end
   else if fail then begin
      SendB(cCAN);
      SendB(cCAN);
      LmxWrLn('SEND FAILED')
   end
   else begin
      LmxWr('sent ');
      LmxNum(fsize);
      LmxWr(' bytes as ');
      LmxNum(nblocks);
      LmxWrLn(' YMODEM blocks')
   end;
   Sched_Class(schedErr, false);
   CLOSE_OBJECT(ecode, frefnum);
   Sched_Class(schedErr, true);
   PortDn
end;

procedure GatherCat;
{ Read the catalog of LISACom's own volume (myVol) into pickName once.  }
{ 1.3: was hard-coded to -#12-, so List/Send/Delete saw the boot disk   }
{ even when the tool and its downloads were on another volume.         }
var path     : Pathname;
    prefix   : E_Name;
    entry    : E_Name;
    schedErr : integer;
begin
   Sched_Class(schedErr, false);
   path := listVol;                    { 1.4: the volume being listed }
   prefix := '';
   pickCnt := 0;
   RESET_CATALOG(ecode, path);
   if ecode <> 0 then begin
      LmxWr('RESET_CATALOG ecode ');
      LmxNum(ecode);
      LmxWrLn('');
      Sched_Class(schedErr, true);
      exit(GatherCat)
   end;
   GET_NEXT_ENTRY(ecode, prefix, entry);
   while (ecode = 0) and (pickCnt < maxPick) do begin
      pickCnt := pickCnt + 1;
      pickName[pickCnt] := entry;
      GET_NEXT_ENTRY(ecode, prefix, entry)
   end;
   Sched_Class(schedErr, true)
end;

procedure ShowPage;
{ Print up to pageLines entries, then either pause for RETURN or, at    }
{ the end of the list, ask for a number (picker) or just stop (List).   }
var shown : integer;
    lines : integer;
    ns    : Str255;
begin
   shown := 0;
   { 1.1: page size follows the window - fill what fits, minus room for }
   { the more-prompt - instead of a fixed 16.                            }
   lines := visLines - 2;
   if lines < 4 then lines := pageLines;
   while (lstPos <= lstTotal) and (shown < lines) do begin
      NumStr(lstPos, ns);
      LmxWr(ns);
      LmxWr('  ');
      LmxWrLn(pickName[lstPos]);
      lstPos := lstPos + 1;
      shown := shown + 1
   end;
   if lstPos <= lstTotal then begin
      uiState := stMoreLst;
      inLine := '';
      if lstDel then
         LmxWrap('-- more: RETURN = next page, a number picks that file, Q stops --')
      else if lstPick then
         LmxWrap('-- more: RETURN = next page, a number sends that file, Q stops --')
      else
         LmxWrap('-- more: RETURN = next page, Q stops --')
   end
   else if lstDel then begin
      uiState := stAskDelN;
      inLine := '';
      LmxWrLn('');
      LmxWr('Delete which number (RETURN cancels): ')
   end
   else if lstPick then begin
      uiState := stAskPick;
      inLine := '';
      LmxWrLn('');
      LmxWrLn('Type the number of the file to send, then RETURN.')
   end
   else begin
      uiState := stIdle;
      LmxWr('-- ');
      LmxNum(lstTotal);
      LmxWrLn(' entries')
   end
end;

procedure ShowFree;
{ One line: free blocks on LISACom's own volume.  A full disk can crash  }
{ machine mid-write, so it is worth a glance before receiving anything.  }
var fsi   : Fs_Info;
    fpath : Pathname;
    fb    : longint;
begin
   fpath := listVol;                   { 1.4: the volume being listed }
   LOOKUP(ecode, fpath, fsi);
   if ecode <> 0 then begin
      LmxWr('(free space unknown, LOOKUP ecode ');
      LmxNum(ecode);
      LmxWrLn(')');
      exit(ShowFree)
   end;
   fb := fsi.freecount;
   LmxWr('Volume ');
   LmxWr(listVol);
   LmxWr(': ');
   LmxNum(fb);
   LmxWr(' free blocks (about ');
   LmxNum(fb div 2);
   LmxWrLn(' KB).')
end;

{------------------------------------------------------------------------}
{ 1.6  Use Volume: find the mounted volumes and make one the working     }
{ volume.  LOOKUP on a device prefix answers for a mounted volume; an    }
{ empty slot or drive returns an error and is skipped.  The tool's own   }
{ volume is always entry 1.                                              }
{------------------------------------------------------------------------}

procedure TryVol (var pfx : E_Name);
var fsi      : Fs_Info;
    ec       : integer;
    schedErr : integer;
    path     : Pathname;
    ns       : Str255;
    dup, k   : integer;
begin
   if volCnt >= maxVols then
      exit(TryVol);
   if pfx = myVol then
      exit(TryVol);                    { already entry 1 }
   path := pfx;
   Sched_Class(schedErr, false);
   LOOKUP(ec, path, fsi);
   Sched_Class(schedErr, true);
   if ec <> 0 then
      exit(TryVol);
   if not fsi.mounted then               { 1.6d: a device with no volume }
      exit(TryVol);                      { answers LOOKUP too (0 free)   }
   dup := 0;
   for k := 1 to volCnt do
      if volFree[k] = fsi.freecount then dup := k;
   volCnt := volCnt + 1;
   volList[volCnt] := pfx;
   volFree[volCnt] := fsi.freecount;
   NumStr(volCnt, ns);
   LmxWr(ns);
   LmxWr('  ');
   LmxWr(pfx);
   LmxWr('  ');
   LmxNum(fsi.freecount);
   LmxWr(' free blocks');
   if dup > 0 then begin               { 1.7: same disk under another name? }
      LmxWr('  (same free count as ');
      NumStr(dup, ns);
      LmxWr(ns);
      LmxWr(' - probably the same disk)')
   end;
   LmxWrLn('')
end;

procedure StartUseVol;
var pfx      : E_Name;
    fsi      : Fs_Info;
    ec       : integer;
    schedErr : integer;
    path     : Pathname;
begin
   LmxWrLn('');
   LmxWrLn('Mounted volumes:');
   volCnt := 1;
   volList[1] := myVol;
   volFree[1] := -1;
   path := myVol;
   Sched_Class(schedErr, false);
   LOOKUP(ec, path, fsi);
   Sched_Class(schedErr, true);
   if ec = 0 then
      volFree[1] := fsi.freecount;
   LmxWr('1  ');
   LmxWr(myVol);
   LmxWr('  ');
   if volFree[1] >= 0 then begin
      LmxNum(volFree[1]);
      LmxWr(' free blocks')
   end;
   LmxWrLn('  (this tool''s volume)');
   pfx := '-#12-';   TryVol(pfx);
   pfx := '-#2#1-';  TryVol(pfx);
   pfx := '-#2#2-';  TryVol(pfx);
   pfx := '-#3#1-';  TryVol(pfx);
   pfx := '-#3#2-';  TryVol(pfx);
   pfx := '-#1#1-';  TryVol(pfx);
   pfx := '-#1#2-';  TryVol(pfx);
   pfx := '-LOWER-'; TryVol(pfx);
   pfx := '-UPPER-'; TryVol(pfx);
   LmxWr('Working volume now: ');
   LmxWrLn(curVol);
   uiState := stAskVol;
   inLine := '';
   LmxWr('Use which number (RETURN cancels): ')
end;

procedure AskDelete (k : integer);

procedure TermEnter;
begin
   cancelled := false;
   if not lineOpen then begin
      if not PortUp then
         exit(TermEnter);
      lineOpen := true               { the terminal owns the line now }
   end;
   uiState := stTermnl;
   escState := 0;
   LmxWrLn('');
   LmxWr('TERMINAL at ');
   LmxNum(curBaud);
   LmxWrLn(' baud.  Apple-. or Apple-Q exits; any menu');
   LmxWrLn('also exits.  The line stays open afterward.');
   LmxWr('Apple-E toggles local echo (now ');
   if termEcho then LmxWr('on)') else LmxWr('off)');
   LmxWr('; Apple-L toggles RETURN = CR+LF (now ');
   if termCRLF then LmxWr('CR+LF)') else LmxWr('CR only)');
   LmxWr('; Apple-B toggles Backspace = BS (now ');
   if termBS then LmxWrLn('BS).') else LmxWrLn('DEL).');
   LmxWrLn('')
end;

procedure TermExit;
begin
   if uiState = stTermnl then begin
      uiState := stIdle;
      LmxWrLn('');
      LmxWrLn('terminal closed - the line is still open.');
      LmxWrLn('Hang Up (XModem menu) closes it.')
   end
end;

procedure TermKey (ch : char);
{ One keystroke out the port.  RETURN goes as CR, or CR+LF when the   }
{ Apple-L toggle is on.  Modems (the WiRSa among them) want a bare CR }
{ and ignore a command that ends CR+LF; a PC terminal program set to  }
{ receive CR only (Tera Term's default) wants the LF, so both are    }
{ offered.                                                            }
var tmp : string[1];
begin
   if (ch = chr(8)) and (not termBS) then
      SendB(127)                      { most BBS software wants DEL }
   else
      SendB(ord(ch));
   if (ch = chr(13)) and termCRLF then
      SendB(10);
   if termEcho then begin
      if ch = chr(13) then
         LmxWrLn('')
      else if ch >= ' ' then begin
         tmp := ' ';
         tmp[1] := ch;
         LmxWrap(tmp)
      end
   end
end;

procedure TermBack;
{ The remote sent BS or DEL: take the last character off the current  }
{ pane line and repaint that row.                                      }
var band : Rect;
begin
   if length(paneBuf[paneCnt]) = 0 then exit(TermBack);
   paneBuf[paneCnt][0] := chr(length(paneBuf[paneCnt]) - 1);
   if paneUp and (myWindow <> nil) then begin
      SetPort(myWindow);
      band := paneRect;
      band.top := paneRect.top + (visRow - 1) * lineHt;
      band.bottom := band.top + lineHt;
      EraseRect(band);
      MoveTo(paneRect.left, band.top + ascentPx);
      DrawString(paneBuf[paneCnt]);
      curX := paneRect.left + StringWidth(paneBuf[paneCnt])
   end
end;

procedure TermPoll;
{ 1.3: model-first terminal drain.  The old path rendered as it        }
{ parsed - one LmxWrLn, meaning one ScrollRect, per received CR - and  }
{ on a 5 MHz 68000 a 2 KB menu burst outran the driver's 1024-byte     }
{ typeahead while the pane scrolled: bytes were silently dropped and   }
{ menus died mid-word (modem flow control is off, and the 8 K buffer   }
{ experiment proved 1024 is a hard limit on the 2/10).  Each poll now  }
{ drains up to 600 bytes into the PANE MODEL only, then paints ONCE:   }
{ a single DrawPane when lines were added or the screen cleared, or a  }
{ one-line repaint when only the current line grew.  Escape handling   }
{ is unchanged from 1.1.  One deliberate behavior change: lines        }
{ longer than 80 columns now TRUNCATE like a real 80-column terminal   }
{ instead of word-wrapping.                                             }
var count  : longint;
    grab   : integer;
    i, b   : integer;
    budget : integer;
    nl     : integer;
    touch  : boolean;
    quiet  : boolean;
    band   : Rect;

   procedure TCh (ch : char);
   { one character into the current model line, 80-column cap }
   var t : string[1];
   begin
      if length(paneBuf[paneCnt]) < paneCols then begin
         t := ' ';
         t[1] := ch;
         paneBuf[paneCnt] := concat(paneBuf[paneCnt], t)
      end;
      touch := true
   end;

   procedure TNl;
   { model-only newline: LmxWrLn's bookkeeping without the painting }
   var j : integer;
   begin
      if paneCnt < paneLines then
         paneCnt := paneCnt + 1
      else
         for j := 1 to paneLines - 1 do
            paneBuf[j] := paneBuf[j + 1];
      paneBuf[paneCnt] := '';
      nl := nl + 1;
      touch := true
   end;

   procedure TClr;
   { model-only clear screen }
   begin
      paneCnt := 1;
      paneBuf[1] := '';
      nl := nl + 1;                      { forces the full repaint }
      touch := true
   end;

begin
   nl := 0;
   touch := false;
   quiet := false;
   budget := 600;
   while (budget > 0) and (not quiet) do begin
      cparm.dccode := 19;
      DEVICE_CONTROL(ecode, pnp, cparm);
      if ecode <> 0 then
         quiet := true
      else begin
         count := cparm.dcdata[9];
         if count <= 0 then
            quiet := true
         else begin
            grab := 128;
            if count < grab then
               grab := count;
            if budget < grab then
               grab := budget;
            READ_DATA(ecode, prefnum, ord4(@pbuf), grab, actual,
                      sequential, 0);
            if (ecode <> 0) or (actual <= 0) then
               quiet := true
            else begin
               budget := budget - actual;
               for i := 0 to actual - 1 do begin
                  b := pbuf[i];
                  if b < 0 then b := b + 256;
                  if escState = 1 then begin          { the byte after ESC }
                     if b = 91 then begin             { '[' : CSI follows }
                        escState := 2;
                        escArg := 0
                     end
                     else if (b = 40) or (b = 41) or (b = 35) then
                        escState := 3                 { '(' ')' '#' : one more }
                     else if (b = 93) or (b = 80) or (b = 88) or (b = 94)
                        or (b = 95) then
                        escState := 4                 { ']' 'P' 'X' '^' '_' }
                     else
                        escState := 0                 { two-byte, done }
                  end
                  else if escState = 2 then begin     { inside CSI }
                     if (b >= 48) and (b <= 57) then
                        escArg := escArg * 10 + (b - 48)
                     else if b = 59 then
                        escArg := 0                   { ';' next parameter }
                     else if (b >= 64) and (b <= 126) then begin
                        escState := 0;                { final byte: done }
                        if (b = 74) and (escArg = 2) then   { ESC [ 2 J }
                           TClr
                     end
                  end
                  else if escState = 3 then
                     escState := 0
                  else if escState = 4 then begin
                     { OSC/DCS string: swallow until BEL or a new ESC }
                     if b = 7 then escState := 0
                     else if b = 27 then escState := 1
                  end
                  else if b = 27 then
                     escState := 1
                  else if b = 13 then
                     TNl
                  else if (b = 8) or (b = 127) then begin
                     if length(paneBuf[paneCnt]) > 0 then begin
                        paneBuf[paneCnt][0] :=
                           chr(length(paneBuf[paneCnt]) - 1);
                        touch := true
                     end
                  end
                  else if b = 12 then                 { form feed: clear }
                     TClr
                  else if (b >= 32) and (b < 127) then
                     TCh(chr(b))
                  else if b >= 128 then
                     TCh(cpTab[b - 127])
                  { LF, BEL and the rest are quietly dropped }
               end
            end
         end
      end
   end;
   { paint exactly once for everything this poll drained }
   if touch and paneUp and (myWindow <> nil) then begin
      SetPort(myWindow);
      if nl > 0 then
         DrawPane
      else begin
         band := paneRect;
         band.top := paneRect.top + (visRow - 1) * lineHt;
         band.bottom := band.top + lineHt;
         EraseRect(band);
         MoveTo(paneRect.left, band.top + ascentPx);
         DrawString(paneBuf[paneCnt]);
         curX := paneRect.left + StringWidth(paneBuf[paneCnt])
      end
   end
end;

{------------------------------------------------------------------------}
{ BBSLIST.TEXT -- bookmarks and defaults.                                }
{                                                                        }
{ A plain text file on the boot volume, one entry per line:              }
{    BAUD 9600            default line speed at launch                   }
{    INIT ATZ             sent to the modem before each dial (optional)  }
{    NAME 8-Bit Boyz|ATDT bbs.8bitboyz.com:23                            }
{ Anything else is ignored, so a typo cannot break the file.  It is       }
{ read once at launch; make it on the PC and send it over, or edit it     }
{ in the Workshop.  A Workshop .TEXT file is 1024-byte pages: this        }
{ reader walks the raw bytes and treats CR as end-of-line, skips NULs.    }
{------------------------------------------------------------------------}

procedure BookAdd (var line : Str255);
{ Parse one line: keyword, space, rest. }
var kw   : Str255;
    rest : Str255;
    i, k : integer;
    bar  : integer;
begin
   i := 1;
   while (i <= length(line)) and (line[i] <> ' ') do i := i + 1;
   kw := copy(line, 1, i - 1);
   while (i <= length(line)) and (line[i] = ' ') do i := i + 1;
   rest := copy(line, i, length(line) - i + 1);
   for k := 1 to length(kw) do
      if kw[k] in ['a'..'z'] then kw[k] := chr(ord(kw[k]) - 32);

   if kw = 'BAUD' then begin
      k := 0;
      for i := 1 to length(rest) do
         if rest[i] in ['0'..'9'] then
            k := k * 10 + (ord(rest[i]) - ord('0'));
      if (k = 1200) or (k = 2400) or (k = 4800) or (k = 9600)
         or (k = 19200) or (k = 28800) or (k = 38400) then
         curBaud := k
   end
   else if kw = 'INIT' then
      initStr := rest
   else if kw = 'NAME' then begin
      bar := pos('|', rest);
      if (bar > 1) and (bookCnt < maxBook) then begin
         bookCnt := bookCnt + 1;
         k := bar - 1;
         if k > bookNmLen then k := bookNmLen;
         bookName[bookCnt] := copy(rest, 1, k);
         rest := copy(rest, bar + 1, length(rest) - bar);
         { optional third field: the note }
         bar := pos('|', rest);
         if bar > 0 then begin
            k := length(rest) - bar;
            if k > bookNtLen then k := bookNtLen;
            bookNote[bookCnt] := copy(rest, bar + 1, k);
            rest := copy(rest, 1, bar - 1)
         end
         else
            bookNote[bookCnt] := '';
         k := length(rest);
         if k > bookCmLen then k := bookCmLen;
         bookCmd[bookCnt] := copy(rest, 1, k)
      end
   end
end;

procedure LoadBooks;
{ Read BBSLIST.TEXT on LISACom's own volume, if it exists.  Absent:     }
{ are optional.                                                          }
var path     : Pathname;
    fref     : integer;
    line     : Str255;
    got      : longint;
    i, b     : integer;
    schedErr : integer;
    tmp      : string[1];
begin
   bookCnt := 0;
   initStr := '';
   bookMsg := 'no BBSLIST.TEXT found (bookmarks are optional)';
   path := concat(myVol, 'BBSLIST.TEXT');   { 1.5: this volume }
   Sched_Class(schedErr, false);
   OPEN(ecode, path, fref, [DRead]);
   Sched_Class(schedErr, true);
   if ecode > 0 then begin
      { 1.2: no file -> one built-in bookmark, The Apple Lisa BBS }
      line := 'NAME LISA BBS|ATDT theapplelisabbs.duckdns.org:1983';
      line := concat(line, '|The Apple Lisa BBS - 9600 baud, protocol Y');
      BookAdd(line);
      bookMsg := 'no BBSLIST.TEXT: built-in bookmark LISA BBS';
      exit(LoadBooks)
   end;
   line := '';
   repeat
      Sched_Class(schedErr, false);
      READ_DATA(ecode, fref, ord4(@pbuf), 128, got, sequential, 0);
      Sched_Class(schedErr, true);
      { end-of-file arrives as a nonzero ecode TOGETHER with the last  }
      { short chunk: use those bytes, then stop.                        }
      if got < 0 then got := 0;
      if got > 128 then got := 128;
      for i := 0 to got - 1 do begin
         b := pbuf[i];
         if b < 0 then b := b + 256;
         if b = 13 then begin
            if length(line) > 0 then BookAdd(line);
            line := ''
         end
         else if (b >= 32) and (b < 127) and (length(line) < 200) then begin
            tmp := ' ';
            tmp[1] := chr(b);
            line := concat(line, tmp)
         end
         { NUL page padding, LF, DLE compression bytes: skipped }
      end
   until (got = 0) or (ecode <> 0);
   if length(line) > 0 then BookAdd(line);
   Sched_Class(schedErr, false);
   CLOSE_OBJECT(ecode, fref);
   Sched_Class(schedErr, true);
   bookMsg := 'BBSLIST.TEXT: ';
   NumStr(bookCnt, line);
   bookMsg := concat(bookMsg, concat(line, ' bookmarks'))
end;

procedure PutLine (fref : integer; var s : Str255);
{ One line out to the bookmark file: the text then CR.  A Workshop      }
{ .TEXT reader treats CR as end-of-line; our own LoadBooks certainly    }
{ does.  (No page formatting: this file is ours, and LoadBooks skips    }
{ NULs anyway, so a plain CR-separated file is fine.)                   }
var w : longint;
    b : array [0..0] of Byte;
    i : integer;
    e : integer;
begin
   for i := 1 to length(s) do begin
      b[0] := ord(s[i]);
      if b[0] > 127 then b[0] := b[0] - 256;
      WRITE_DATA(e, fref, ord4(@b), 1, w, sequential, 0)
   end;
   b[0] := 13;
   WRITE_DATA(e, fref, ord4(@b), 1, w, sequential, 0)
end;

procedure SaveBooks;
{ Rewrite BBSLIST.TEXT from what is in memory.  BAUD and INIT survive.  }
var path     : Pathname;
    fref     : integer;
    line     : Str255;
    i        : integer;
    schedErr : integer;
    e        : integer;
begin
   path := concat(myVol, 'BBSLIST.TEXT');   { 1.5: this volume }
   Sched_Class(schedErr, false);
   MAKE_FILE(e, path, 0);              { 890 = exists, fine }
   OPEN(e, path, fref, [DWrite]);
   Sched_Class(schedErr, true);
   if e > 0 then begin
      LmxWr('cannot write BBSLIST.TEXT, ecode ');
      LmxNum(e);
      LmxWrLn('');
      exit(SaveBooks)
   end;
   NumStr(curBaud, line);
   line := concat('BAUD ', line);
   PutLine(fref, line);
   if length(initStr) > 0 then begin
      line := concat('INIT ', initStr);
      PutLine(fref, line)
   end;
   for i := 1 to bookCnt do begin
      line := concat('NAME ', bookName[i]);
      line := concat(line, concat('|', bookCmd[i]));
      if length(bookNote[i]) > 0 then
         line := concat(line, concat('|', bookNote[i]));
      PutLine(fref, line)
   end;
   Sched_Class(schedErr, false);
   TRUNCATE(e, fref);
   CLOSE_OBJECT(e, fref);
   Sched_Class(schedErr, true);
   NumStr(bookCnt, line);
   LmxWr('BBSLIST.TEXT saved: ');
   LmxWr(line);
   LmxWrLn(' bookmarks.')
end;

procedure SendStr (var s : Str255);
{ A whole string out the port, then CR.  Modem commands end in CR    }
{ alone; a trailing LF makes the WiRSa drop the command.             }
var i : integer;
begin
   for i := 1 to length(s) do
      SendB(ord(s[i]));
   SendB(13)
end;

function WaitOK : boolean;
{ Beta 23; extended beta 24.  After INIT, hold until the modem answers  }
{ before firing the dial string at it.  Beta 24: ERROR now also counts  }
{ as an answer - the WiRSa replies ERROR to ATZ, and beta 23 sat        }
{ through a full silence timeout before dialing.  Any final result      }
{ ends the wait at once; only a true OK returns true, so the caller's   }
{ "(no OK from modem -- dialing anyway)" still prints on ERROR.         }
var b, n, seenO, seenE : integer;
    done, okAns : boolean;
    tmp         : string[1];
begin
   done  := false;
   okAns := false;
   n     := 0;
   seenO := 0;
   seenE := 0;
   while (not done) and (n < 200) and (not cancelled) do begin
      if not GetB(b) then
         n := 200                      { silence - stop waiting }
      else begin
         n := n + 1;
         if (b >= 32) and (b < 127) then begin
            tmp := ' ';
            tmp[1] := chr(b);
            LmxWr(tmp)
         end
         else if b = 13 then
            LmxWrLn('');
         if (b = ord('O')) or (b = ord('o')) then begin
            seenO := 1;
            seenE := 0
         end
         else if ((b = ord('K')) or (b = ord('k'))) and (seenO = 1) then begin
            done := true;
            okAns := true
         end
         else if (b = ord('E')) or (b = ord('e')) then begin
            seenE := 1;
            seenO := 0
         end
         else if ((b = ord('R')) or (b = ord('r'))) and (seenE = 1) then
            done := true               { ERROR - answered, but not OK }
         else begin
            seenO := 0;
            seenE := 0
         end
      end
   end;
   WaitOK := okAns
end;

procedure DialBook (n : integer);
{ Open the line, send INIT then the bookmark's command, land in the     }
{ terminal already connected.                                            }
var dialTmp : Str255;        { VAR params want the exact named type }
begin
   if (n < 1) or (n > bookCnt) then
      exit(DialBook);
   LmxWrLn('');
   LmxWr('Dialing ');
   LmxWrLn(bookName[n]);
   if lineOpen then begin              { 1.2: never dial into an open }
      LmxWrLn('hanging up the open line first.');   { session      }
      HangUp
   end;
   TermEnter;                          { opens the line if needed }
   if uiState <> stTermnl then
      exit(DialBook);                  { the port would not open }
   if length(initStr) > 0 then begin
      SendStr(initStr);
      LmxWr('> ');
      LmxWrLn(initStr);
      if not WaitOK then
         LmxWrLn('(no OK from modem -- dialing anyway)')
   end;
   dialTmp := bookCmd[n];
   SendStr(dialTmp);
   LmxWr('> ');
   LmxWrLn(dialTmp)
end;

function PickedNum : integer;
{ Read inLine as a decimal number; 0 means "not a usable number".        }
var i, v, d : integer;

   v := 0;
   for i := 1 to length(inLine) do begin
      d := ord(inLine[i]) - ord('0');

      end;

   end;

end;

procedure PickVol;
{ 1.6: the answer to Use Volume. }
var k : integer;
begin
   k := PickedNum;
   if (k < 1) or (k > volCnt) then begin
      LmxWrLn('that is not one of the numbers listed.');
      exit(PickVol)
   end;
   curVol := volList[k];
   LmxWr('Working volume is now ');
   LmxWr(curVol);
   LmxWrLn(' - receives, List, browse and Delete use it.')
end;

procedure StartPrompt (which : integer);
begin
   uiState := which;
   inLine := '';
   LmxWrLn('');
   if which = stAskRecv then begin
      LmxWrLn('Receive: a new file here holds what the PC sends.');
      LmxWr('New file name: ')
   end
   else begin
      LmxWrLn('Send: a file on this Lisa will be transmitted to the PC.');
      LmxWr('File to send: ')
   end
end;

procedure PromptBack;
{ Erase the last typed character: shrink the accumulator AND the pane's }
{ current line, then repaint that one row from the buffer.               }
var band : Rect;
begin
   if length(inLine) = 0 then exit(PromptBack);
   inLine[0] := chr(length(inLine) - 1);
   paneBuf[paneCnt][0] := chr(length(paneBuf[paneCnt]) - 1);
   SetPort(myWindow);
   band := paneRect;
   band.top := paneRect.top + (visRow - 1) * lineHt;
   band.bottom := band.top + lineHt;
   EraseRect(band);
   MoveTo(paneRect.left, band.top + ascentPx);
   DrawString(paneBuf[paneCnt]);
   curX := paneRect.left + StringWidth(paneBuf[paneCnt])
end;

function PromptCap : integer;
{ How many characters the current prompt accepts.  The bookmark prompts }
{ stop at their storage sizes so nothing is silently cut off.           }
begin
   if uiState = stBkName then PromptCap := bookNmLen
   else if uiState = stBkCmd then PromptCap := bookCmLen
   else if uiState = stBkNote then PromptCap := bookNtLen
   else PromptCap := 60
end;

procedure PromptKey (ch : char);
var tmp    : string[1];
    which  : integer;
    k      : integer;
    chosen : Str255;
begin
   if uiState = stMoreLst then begin
      { RETURN pages on; a digit starts a number to send; anything else }
      { stops.  Paging through hundreds of files to reach the prompt    }
      { was unusable.                                                    }
      if ch = chr(13) then begin
         uiState := stIdle;
         LmxWrLn('');
         ShowPage
      end
      else if (ch >= '0') and (ch <= '9') and lstDel then begin
         uiState := stAskDelN;
         inLine := '';
         LmxWrLn('');
         LmxWr('Delete file number: ');
         PromptKey(ch)
      end
      else if (ch >= '0') and (ch <= '9') and lstPick then begin
         uiState := stAskPick;
         inLine := '';
         LmxWrLn('');
         LmxWr('Send file number: ');
         PromptKey(ch)          { the digit just typed starts the number }
      end
      else begin
         uiState := stIdle;
         LmxWrLn('');
         LmxWrLn('listing stopped.')
      end;
      exit(PromptKey)
   end;

   if ch = chr(13) then begin
      which := uiState;
      uiState := stIdle;
      LmxWrLn('');
      if (length(inLine) = 0) and (which = stAskName) then begin
         listVol := curVol;             { RETURN alone: working volume }
         StartPick
      end
      else if which = stAskVol then
         PickVol                        { 1.6; RETURN alone = cancelled }
      else if (which = stAskName) and (length(inLine) >= 3)
         and (length(inLine) <= 32)
         and (inLine[1] = '-') and (inLine[length(inLine)] = '-') then begin
         listVol := inLine;             { 1.4: -vol- alone browses it }
         StartPick
      end
      else if which = stBkNote then
         FinishBkAdd(inLine)            { empty note is fine }
      else if which = stYSave then
         DoRecvYOp(inLine)              { empty = take the sender's name }
      else if length(inLine) = 0 then
         LmxWrLn('cancelled.')
      else if which = stAskDial then begin
         k := PickedNum;
         if (k < 1) or (k > bookCnt) then
            LmxWrLn('that is not one of the numbers listed.')
         else
            DialBook(k)
      end
      else if which = stBkName then begin
         k := length(inLine);
         if k > bookNmLen then k := bookNmLen;
         newName := copy(inLine, 1, k);
         uiState := stBkCmd;
         inLine := '';
         LmxWr('Dial command, e.g. ATDT host:port (60 max): ')
      end
      else if which = stBkCmd then begin
         k := length(inLine);
         if k > bookCmLen then k := bookCmLen;
         newCmd := copy(inLine, 1, k);
         uiState := stBkNote;
         inLine := '';
         LmxWr('Note, optional (40 max, RETURN for none): ')
      end
      else if which = stBkDel then begin
         k := PickedNum;
         DoBkDel(k)
      end
      else if which = stAskDelN then begin
         k := PickedNum;
         AskDelete(k)
      end
      else if which = stConfDel then
         DoDelete
      else if which = stAskName then begin
         { a typed name goes straight out; RETURN alone browses }
         if ySend then DoSendYOp(inLine)
         else DoSendOp(inLine)
      end
      else if which = stAskPick then begin
         k := PickedNum;
         if (k < 1) or (k > pickCnt) then
            LmxWrLn('that is not one of the numbers listed.')
         else begin
            chosen := concat(listVol, pickName[k]);  { 1.4: full path }
            LmxWr('sending ');
            LmxWrLn(chosen);
            if ySend then DoSendYOp(chosen)
            else DoSendOp(chosen)
         end
      end
      else begin
         if which = stAskRecv then DoRecvOp(inLine)
         else if ySend then DoSendYOp(inLine)
         else DoSendOp(inLine)
      end;
      if paneUp and (myWindow <> nil) then
         DrawPane                       { resync after the operation }
   end
   else if ch = chr(8) then
      PromptBack
   else if (ch >= ' ') and (length(inLine) < PromptCap) then begin
      inLine := concat(inLine, ' ');
      inLine[length(inLine)] := ch;
      tmp := ' ';
      tmp[1] := ch;
      LmxWr(tmp)
   end
end;

   myWindow := window;
   TakeWindow(myWindow);

   if (paneCnt = 1) and (length(paneBuf[1]) = 0) then begin
      LmxWr('LISACom  -  ');

      LmxWr('Serial B, ');

      LmxWrLn(' 8N1, XMODEM + YMODEM receive.');
      LmxWrLn('Bugs and feature ideas: post to LisaList2.');
      LmxWrLn('Any key stops a transfer in progress.');
      if paneMono then begin
         LmxWr('Pane font ');
         LmxNum(paneFont);
         LmxWr(', fixed width, ');
         LmxNum((paneRect.right - paneRect.left) div paneCW);

   doUpdateEvent(false, myWindow);

   if activeFolder = myWindow then
      PutUpMenus
end;

procedure MenuCommand (menu, item : integer);
var err : integer;
begin
   if uiState = stTermnl then
      TermExit                          { leaves the line open }
   else if uiState <> stIdle then begin { a menu pull cancels prompt/paging }
      uiState := stIdle;
      LmxWrLn('');
      LmxWrLn('cancelled.')
   end;

   case menu of

      mFile :
         case item of
            miAsideAll :
               DoFilingCmd(cmdClosAll);
            miAside :
               DoFilingCmd(cmdClose);
            miPutAway :
               begin
                  TellFiler(err, docClosd, docPutBack, myWindow);
                  myWindow := nil;
                  paneUp := false
               end;
         end;

      mXfer :
         case item of
            miReceive :
               StartPrompt(stAskRecv);
            miRecvY :
               begin
                  uiState := stYSave;
                  inLine := '';
                  LmxWrLn('');
                  LmxWrLn('YMODEM receive: the sender supplies the name.');
                  LmxWr('Save as (RETURN = sender name, or -vol-name): ')
               end;
            miSend :
               begin
                  ySend := false;
                  uiState := stAskName;
                  inLine := '';
                  LmxWrLn('');
                  LmxWrLn('Send: a file on this Lisa goes to the PC.');
                  LmxWr('File to send (RETURN or -vol- to browse): ')
               end;
            miSendY :
               begin
                  ySend := true;
                  uiState := stAskName;
                  inLine := '';
                  LmxWrLn('');
                  LmxWrLn('Send via YMODEM: name and exact size go');
                  LmxWrLn('with the file.');
                  LmxWr('File to send (RETURN or -vol- to browse): ')
               end;
            miList :
               DoListOp;
            miTermnl :
               TermEnter;
            miDial :
               StartDial;
            miHangUp :
               HangUp;
            miUseVol :
               StartUseVol;
            miDelete :
               StartDelete;
            miClear :
               LmxClear;
         end;

      mBook :
         case item of
            miBkAdd  : StartBkAdd;
            miBkDel  : StartBkDel;
            miBkList : ShowBooks;
         end;

      mVer :
       if item = miFonts then
          FontSurvey
       else
         begin
            LmxWr('LISACom  ');
            LmxWr(lmxVersion);
            LmxWrLn('  (was LISA XModem Util, tool 419306)');
            LmxWrLn('Bugs and ideas: post to LisaList2.');
            LmxWrLn('');
            LmxWrLn('If the port will not open, Serial B is not set up');
            LmxWrLn('on this boot volume: quit, open Preferences,');
            LmxWrLn('choose Connect Devices, and set Serial B');
            LmxWrLn('Connector to Serial Cable.');
            LmxWrLn('');
            LmxWrLn('Modem cabling: a WiFi modem is a modem, so use a');
            LmxWrLn('straight cable - no null modem, no Apple modem');
            LmxWrLn('eliminator.  No reply to AT?  Unplug the modem and');
            LmxWrLn('bridge DB9 pins 2-3 at that end: typed keys should');
            LmxWrLn('echo back.  If they do, the fault is the modem side.');
            LmxWrLn('');
            LmxWrLn('Any key stops a transfer in progress.');
            LmxWrLn(bookMsg)
         end;

      mBaud :
         begin
            case item of
               miB1200  : curBaud := 1200;
               miB2400  : curBaud := 2400;
               miB4800  : curBaud := 4800;
               miB9600  : curBaud := 9600;
               miB19200 : curBaud := 19200;
               miB28800 : curBaud := 28800;
               miB38400 : curBaud := 38400;
            end;
            LmxWr('Line speed is now ');
            LmxNum(curBaud);
            LmxWrLn(' baud - match it on the PC.')
         end;

   end;

   { Resync the view from the buffer after every command: menus and     }
   { alerts restore saved pixels over whatever we drew meanwhile.       }
   if paneUp and (myWindow <> nil) then
      DrawPane
end;

procedure ProcessTheEvent{FORWARD above: events reach here even from
inside a transfer, via UserAborted};
var theMenu, theItem : integer;
begin
   case theEvent.what of

      buttonDown :
         if theEvent.who = menuFolder then begin
            MenuSelect(theEvent.where, theMenu, theItem);
            if (theMenu <> 0) and (theItem <> 0) then
               MenuCommand(theMenu, theItem);
            HiLiteMenu(0)
         end;

      buttonUp :
         ;

      keyDown :
         if (theEvent.who = myWindow) and
            ((not theEvent.repeatKey) or (uiState = stTermnl)) then begin
            { auto-repeat is honoured ONLY in the terminal, where a held }
            { key should stream out the port like a real terminal; the  }
            { menus and prompts still ignore repeats.                    }
            if uiState = stTermnl then begin
               { the terminal owns the keyboard: plain keys go out the  }
               { port; Apple-. exits; Apple-E toggles echo; every other }
               { Apple combination is ignored so a stray shortcut       }
               { cannot fire a menu command mid-session.                }
               if theEvent.appleKey then begin
                  if (theEvent.ascii = '.') or (theEvent.ascii = 'Q')
                     or (theEvent.ascii = 'q') then
                     TermExit
                  else if (theEvent.ascii = 'E') or (theEvent.ascii = 'e')
                  then begin
                     termEcho := not termEcho;
                     if termEcho then
                        LmxWrLn(' [echo on]')
                     else
                        LmxWrLn(' [echo off]')
                  end
                  else if (theEvent.ascii = 'L') or (theEvent.ascii = 'l')
                  then begin
                     termCRLF := not termCRLF;
                     if termCRLF then
                        LmxWrLn(' [RETURN sends CR+LF]')
                     else
                        LmxWrLn(' [RETURN sends CR only]')
                  end
                  else if (theEvent.ascii = 'B') or (theEvent.ascii = 'b')
                  then begin
                     termBS := not termBS;
                     if termBS then
                        LmxWrLn(' [Backspace sends BS]')
                     else
                        LmxWrLn(' [Backspace sends DEL]')
                  end
               end
               else
                  TermKey(theEvent.ascii)
            end
            else if theEvent.appleKey then begin
               MenuKey(theEvent.ascii, theMenu, theItem);
               if theMenu <> 0 then begin
                  HiLiteMenu(theMenu);
                  MenuCommand(theMenu, theItem);
                  HiLiteMenu(0)
               end
            end
            else if uiState <> stIdle then
               PromptKey(theEvent.ascii)
         end;

      folderActivate :
         begin
            TakeControl(theEvent, false, false);
            PutUpMenus
         end;

      folderDeactivate :
         GiveControl(theEvent);

      folderUpdate :
         if theEvent.who = myWindow then
            doUpdateEvent(true, theEvent.who)
         else begin
            BeginUpdate(theEvent.who);
            EndUpdate(theEvent.who)
         end;

      folderMoved :
         ;

      filerEvent :
         DoFilerEvent;

      abortEvent, diedEvent, diskEvent, catalogEvent, nilEvent :
         ;

      otherwise

   end
end;

procedure Initialize;
var err      : integer;
    schedErr : integer;
    pcInfo   : procinfoRec;
    excName  : t_ex_name;
    myPrefix : TParamAlert;
    cutAt    : integer;
    tmp1     : string[1];
begin
   initing := true;

   excName := 'SYS_TERMINATE';
   Declare_Excep_Hdl(err, excName, @LastWishes);
   if err <> 0 then begin
      TellFiler(err, initFailed, noMemory, nil);
      halt
   end;

   toldToDie := false;
   myWindow  := nil;
   paneUp    := false;
   paneCnt   := 0;
   scrollRgn := nil;
   uiState   := stIdle;
   cancelled := false;
   lineOpen  := false;
   termEcho  := false;
   termCRLF  := false;
   termBS    := false;
   crcTabOK  := false;
   escState  := 0;
   cpTab := cpMap;                     { 1.1: see cpTab declaration }
   escArg    := 0;
   curBaud   := 9600;
   dotsLine  := '';
   pickCnt   := 0;

   InitHeap(@myZone[0], @myZone[maxHeap], @OutOfHeapSpace);

   Sched_Class(schedErr, false);
   Info_Process(err, My_ID, pcInfo);
   Sched_Class(schedErr, true);

   cutAt := pos('}', pcInfo.progpathname);
   if (cutAt = 0) or (cutAt > 40) then begin
      TellFiler(err, initFailed, internalError, nil);
      halt
   end;
   myPrefix := copy(pcInfo.progpathname, 1, cutAt);
   { 1.2: the device prefix alone (before the tool's open brace) is    }
   { where received files go.  chr(123) is the brace, SanName-style.   }
   tmp1 := ' ';
   tmp1[1] := chr(123);
   cutAt := pos(tmp1, pcInfo.progpathname);
   if cutAt > 1 then
      myVol := copy(pcInfo.progpathname, 1, cutAt - 1)
   else
      myVol := '-#12-';                { defensive fallback: boot }
   listVol := myVol;                   { 1.4 }
   curVol := myVol;                    { 1.6: working volume starts here }
   volCnt := 0;

   InitMP(theHeap, myPrefix, @AlertFailed, alertVerNum,
          @myMenus[mFile], allAlerts, err);
   if err > 0 then begin
      TellFiler(err, initFailed, noMemory, nil);
      halt
   end;

   InitWmlCrs(err);
   if err <> 0 then begin
      TellFiler(err, initFailed, noDiskSpace, nil);
      halt
   end;

   SetPort(deskPort);

   LoadBooks;                 { BBSLIST.TEXT: baud, init string, bookmarks }

   initing := false
end;

{------------------------------------------------------------------------}
{ The main program block -- the two-phase loop, unchanged.               }
{------------------------------------------------------------------------}

begin { main program }

   InitGraf(@thePort);
   OpenWm;
   Initialize;

   repeat

      while not (ImActive or toldToDie) do begin
         GetEvent(theEvent);
         ProcessTheEvent
      end;

      while ImActive and (not toldToDie) do begin
         if EventAvail then begin
            GetEvent(theEvent);
            ProcessTheEvent
         end
         else begin
            { idle: the terminal drinks from the port here, so incoming }
            { bytes appear while the desktop stays fully alive.         }
            if (uiState = stTermnl) and lineOpen then
               TermPoll;
            LetOthersRun
         end
      end;

   until toldToDie;

   termErr := 0
end.
